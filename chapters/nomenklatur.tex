\ifenglish
\section*{Latin symbols}
\else
\section*{Lateinische Symbole}
\fi
\begin{longtable}[t]{@{}p{0.16\textwidth}p{0.80\textwidth}@{}}
    \textbf{Symbol}  & \textbf{Description}\\
    \midrule
    \endhead
    $B$ & additive constant of the logarithmic law of the wall \\
    $C_p$ & pressure coefficient \\
    $D_{ij}$ & viscous diffusion rate (Reynolds-stress budget) \\
    $k$ & turbulent kinetic energy \\
    $L$ & characteristic length scale \\
    $L_x,\,L_y,\,L_z$ & domain size in streamwise, wall-normal and spanwise direction \\
    $N$ & polynomial order of a spectral element \\
    $N_{\mathrm{eval}}$ & number of cells inside the optimisation evaluation window \\
    $n$ & wall-normal coordinate (curvilinear/moving frame) \\
    $p$ & pressure \\
    $p'$ & pressure fluctuation / pressure-correction variable \\
    $p_0$ & free-stream pressure \\
    $p_w$ & wall pressure \\
    $\langle p_c\rangle$ & mean wall pressure in the curved section \\
    $\mathcal{P}_{ij}$ & production rate (Reynolds-stress budget) \\
    $R$ & residual of the optimisation; curvature radius \\
    $R_{ij}$ & Reynolds-stress tensor \\
    $r$ & physical curvature radius \\
    $s$ & streamwise coordinate (curvilinear/moving frame) \\
    $s_{c}$ & arc length of the curved section \\
    $s_{in},\,s_{out}$ & length of the straight inlet and straight outlet \\
    $T_{ij}$ & turbulent transport rate (Reynolds-stress budget) \\
    $t$ & time \\
    $U,\,V,\,W$ & mean velocity components \\
    $U_0$ & free-stream (reference) velocity \\
    $U_b$ & bulk velocity \\
    $U(s)$ & local free-stream velocity \\
    $u,\,v,\,w$ & velocity components in streamwise, wall-normal and spanwise direction \\
    $u_i$ & velocity component (index notation) \\
    $u_i'$ & fluctuating velocity component \\
    $u_\tau$ & friction velocity \\
    $u^+$ & velocity in viscous (wall) units \\
    $x,\,y,\,z$ & Cartesian coordinates (streamwise, wall-normal, spanwise) \\
    $x_{\theta_i},\,y_{\theta_i}$ & coordinates of design point $\theta_i$ \\
    $y^+$ & wall-normal coordinate in viscous (wall) units \\
\end{longtable}
